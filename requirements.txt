def build_query(card_set: str, character: str, rarity: str, card_no: str, extra: str) -> str:
    parts = ["Naruto Kayou"]
    if card_set and not card_set.startswith("All"):
        parts.append(card_set)
    if character:
        parts.append(character.strip())
    if rarity and rarity != "All":
        parts.append(rarity.strip())
    if card_no:
        parts.append(card_no.strip())
    if extra:
        parts.append(extra.strip())
    return " ".join([p for p in parts if p])


def ebay_active_url(query: str) -> str:
    return f"https://www.ebay.com/sch/i.html?_nkw={quote_plus(query)}"


def ebay_sold_url(query: str) -> str:
    return f"https://www.ebay.com/sch/i.html?_nkw={quote_plus(query)}&LH_Sold=1&LH_Complete=1"


def point130_url(query: str) -> str:
    return f"https://130point.com/sales/?search={quote_plus(query)}"


def clean_number(value):
    if pd.isna(value):
        return None
    text = str(value)
    text = re.sub(r"[^0-9.\-]", "", text)
    try:
        return float(text)
    except Exception:
        return None


def normalize_manual_csv(uploaded_file, fx_rate: float) -> pd.DataFrame:
    df = pd.read_csv(uploaded_file)
    lower_map = {c.lower().strip(): c for c in df.columns}

    def pick(*names):
        for name in names:
            if name.lower() in lower_map:
                return lower_map[name.lower()]
        return None

    title_col = pick("title", "name", "card", "item")
    price_col = pick("price", "sold_price", "usd_price", "price_usd")
    url_col = pick("url", "link", "item_url")
    img_col = pick("image_url", "image", "img", "picture")

    out = pd.DataFrame()
    out["title"] = df[title_col].astype(str) if title_col else "Manual item"
    out["price_usd"] = df[price_col].apply(clean_number) if price_col else None
    out["price_thb"] = out["price_usd"] * fx_rate
    out["item_url"] = df[url_col].astype(str) if url_col else ""
    out["image_url"] = df[img_col].astype(str) if img_col else ""
    out["source"] = "Manual CSV"
    return out.dropna(subset=["price_usd"])


def remove_outliers_iqr(df: pd.DataFrame, col: str = "price_thb") -> pd.DataFrame:
    if df.empty or col not in df or len(df) < 4:
        return df
    q1 = df[col].quantile(0.25)
    q3 = df[col].quantile(0.75)
    iqr = q3 - q1
    low = q1 - 1.5 * iqr
    high = q3 + 1.5 * iqr
    return df[(df[col] >= low) & (df[col] <= high)]


def fetch_ebay_api(query: str, token: str, limit: int, fx_rate: float) -> pd.DataFrame:
    if not token:
        return pd.DataFrame()
    endpoint = "https://api.ebay.com/buy/browse/v1/item_summary/search"
    headers = {"Authorization": f"Bearer {token}", "X-EBAY-C-MARKETPLACE-ID": "EBAY_US"}
    params = {"q": query, "limit": min(max(int(limit), 1), 200)}
    res = requests.get(endpoint, headers=headers, params=params, timeout=25)
    res.raise_for_status()
    items = res.json().get("itemSummaries", [])
    rows = []
    for item in items:
        price = item.get("price", {}) or {}
        image = item.get("image", {}) or {}
        try:
            usd = float(price.get("value"))
        except Exception:
            continue
        rows.append({
            "title": item.get("title", ""),
            "price_usd": usd,
            "price_thb": usd * fx_rate,
            "item_url": item.get("itemWebUrl", ""),
            "image_url": image.get("imageUrl", ""),
            "source": "eBay Active API",
        })
    return pd.DataFrame(rows)


def to_excel_bytes(df: pd.DataFrame, summary: pd.DataFrame) -> bytes:
    buffer = io.BytesIO()
    with pd.ExcelWriter(buffer, engine="openpyxl") as writer:
        df.to_excel(writer, index=False, sheet_name="Price Results")
        summary.to_excel(writer, index=False, sheet_name="Summary")
    return buffer.getvalue()
